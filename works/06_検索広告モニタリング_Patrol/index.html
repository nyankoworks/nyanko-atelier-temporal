検索広告モニタリング「Patrol」

検索広告（スポンサー枠）を自動で見回り、競合の出稿状況をデータ化する自作ツール。

PythonPlaywright 競合分析業務自動化
💡 本当はこのページ上で実際に動かせるようにしたかったのですが、裏側でブラウザを操作する必要があり、無料の静的サイトでは動かせません。そこで「なぜ作ったか・こだわり・工程・コード」を公開します。

🐾 なぜ作ったのか

自社の「指名キーワード」（ブランド名での検索）に、競合他社が広告を出して見込み客を横取りしていないか——これを継続的にチェックするために作りました。指名キーワードは、もっとも購入意欲の高い人がたどる大切な入口です。そこに競合が入札してくると、本来こちらに来るはずのお客様を取られ、機会損失につながります。

ただ、いつ・どの競合が出稿しているかは日によって変わるため、毎回手で検索して記録するのは大変です。そこで「誰が・どんな広告文で・どれくらいの頻度で出しているか」を自動で定点観測し、証拠スクリーンショットと一緒に記録できるようにしました。

🐾 こだわったこと

① 「素の」広告を見る
シークレット（プライベート）モードで巡回し、自分の検索履歴に影響されない、フラットな状態の広告を取得します。
② 証拠を残す
各ページを全画面スクリーンショットで保存。「いつ・どんな広告が出ていたか」を後から確認できます。
③ 表面で終わらせない
広告のリンク先LPまで自動でたどり、会社名・電話番号・問い合わせフォームのURLまで抽出します。
④ 時系列で変化を追う
実行するたびに日時フォルダへ保存し、広告主ごとの「出稿回数」を集計。競合の力の入れ具合の変化が見えます。
⑤ 費用対効果を可視化
「手作業なら何分かかるか × 時給」で、削減できた人件費を自動計算して表示します。

🐾 作り方（工程）

  1. 要件整理：何を・どこから・どう残すかを決める（広告枠／証拠スクショ／CSV）
  2. 巡回：Playwrightで検索し、複数ページを自動で移動
  3. 抽出：広告（スポンサー枠）だけを判定し、タイトル・説明・遷移先を取得
  4. 深掘り：遷移先LPを開いて会社名・連絡先・問い合わせURLを解析
  5. 保存：全画面スクショ＋CSVを日時フォルダに出力
  6. 集計：広告主ごとの出稿回数をサマリー化
  7. 改善：実行プログラム自体もフォルダに自動バックアップして再現性を確保

🐾 コード（実際のスクリプト）

実際に動かしているPython + Playwrightのスクリプトです（検索キーワードは自由に入力できます。※競合名は一般的な表現に置き換えて掲載）。

import asyncio
from playwright.async_api import async_playwright
import pandas as pd
import datetime
import os
import shutil
import re

# --- 設定 ---
KEYWORD = "調べたいキーワード"  # ← 自由に入力して使う
MAX_PAGES = 3
HOURLY_RATE = 2500
MANUAL_TASK_MINUTES = 80  # フォームのURLまで手動で探すと非常に時間がかかるため、1時間20分に設定

async def get_page_info(context, url):
    """遷移先LPから会社名、連絡先、問い合わせ関連URLを取得する"""
    if not url or not url.startswith("http"):
        return "N/A", "N/A", "N/A", url
    try:
        # プライベートコンテキスト内で新しいページを開く
        temp_page = await context.new_page()
        await temp_page.goto(url, wait_until="load", timeout=20000)
        await temp_page.wait_for_timeout(3000)
        
        real_url = temp_page.url
        content = await temp_page.content()
        
        # 1. 連絡先（電話番号）
        phone_match = re.search(r'0\d{1,4}-\d{1,4}-\d{4}|0120-\d{3}-\d{3}', content)
        phone = phone_match.group() if phone_match else "未検出"
        
        # 2. 会社名の推測
        title_tag = await temp_page.title()
        co_match = re.search(r'([A-Zァ-ヶ一-龠]+株式会社|株式会社[A-Zァ-ヶ一-龠]+)', content)
        company = co_match.group() if co_match else title_tag.split('｜')[0].split('|')[0]

        # 3. 問い合わせURLの探索（リンクテキストに「問合せ」「コンタクト」「フォーム」が含まれるものを抽出）
        contact_url = "未検出"
        links = await temp_page.query_selector_all("a")
        for link in links:
            text = await link.inner_text()
            href = await link.get_attribute("href")
            if href and any(x in text for x in ["問合せ", "お問い合わせ", "コンタクト", "入力フォーム", "相談"]):
                # 相対パスを絶対パスに変換
                if href.startswith("/"):
                    from urllib.parse import urljoin
                    contact_url = urljoin(real_url, href)
                else:
                    contact_url = href
                break

        await temp_page.close()
        return company.strip(), phone, contact_url, real_url
    except:
        return "取得失敗", "取得失敗", "取得失敗", url

async def patrol():
    timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M")
    base_dir = f"{timestamp}_{KEYWORD}"
    img_dir = os.path.join(base_dir, "captures")
    os.makedirs(img_dir, exist_ok=True)

    shutil.copy("patrol.py", os.path.join(base_dir, "patrol_script_backup.py"))

    async with async_playwright() as p:
        # ブラウザ起動
        browser = await p.chromium.launch(headless=False) 
        
        # ★シークレットモード（プライベートウィンドウ）相当のコンテキストを作成
        context = await browser.new_context(
            user_agent="Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
            viewport={'width': 1280, 'height': 1024},
            # 履歴やキャッシュを残さない設定
            java_script_enabled=True
        )
        
        page = await context.new_page()
        results = []
        print(f"--- Yahoo!高度パトロール（プライベートモード / 問い合わせURL抽出） ---")
        
        await page.goto(f"https://search.yahoo.co.jp/search?p={KEYWORD}", wait_until="networkidle")
        
        for i in range(1, MAX_PAGES + 1):
            print(f"{i}ページ目を巡回中...")
            await page.wait_for_timeout(5000) 
            
            shot_path = os.path.join(img_dir, f"Yahoo_p{i}.png")
            await page.screenshot(path=shot_path, full_page=True)
            
            items = await page.query_selector_all(".sw-Card")
            
            for item in items:
                label_el = await item.query_selector(".sw-Ad__label, .sw-Ad")
                label_text = await label_el.inner_text() if label_el else ""
                
                # スポンサーまたは広告ラベルがあるもののみ抽出
                if not label_text:
                    continue

                title_el = await item.query_selector("h3")
                desc_el = await item.query_selector(".sw-Card__description, .sw-Ad__description")
                link_el = await item.query_selector("a")
                
                if title_el and link_el:
                    ad_title = await title_el.inner_text()
                    ad_description = await desc_el.inner_text() if desc_el else "（説明文なし）"
                    raw_url = await link_el.get_attribute("href")
                    
                    print(f"  [{label_text}] 解析中: {ad_title[:15]}...")
                    
                    # 遷移先から高度な情報を取得
                    company, phone, contact_url, real_url = await get_page_info(context, raw_url)
                    
                    results.append({
                        "取得日時": timestamp,
                        "ページ": i,
                        "スポンサー/広告表示": label_text,
                        "広告タイトル": ad_title,
                        "広告説明文": ad_description,
                        "会社名": company,
                        "連絡先電話番号": phone,
                        "問い合わせ先URL": contact_url,
                        "本物のLPリンクURL": real_url,
                        "証拠画像": shot_path
                    })
            
            if i < MAX_PAGES:
                next_btn = await page.query_selector(".sw-Pagination__next, a:has-text('次へ')")
                if next_btn: await next_btn.click()
                else: break

        df = pd.DataFrame(results)
        csv_path = os.path.join(base_dir, f"report_{KEYWORD}.csv")
        df.to_csv(csv_path, index=False, encoding="utf-8-sig")
        
        saved_cost = (MANUAL_TASK_MINUTES / 60) * HOURLY_RATE
        print("\n" + "="*40)
        print(f"【パトロール完了】")
        print(f"作成フォルダ: {base_dir}")
        print(f"本日削減できた人件費: 約 {int(saved_cost)} 円")
        print("="*40)

        await browser.close()

if __name__ == "__main__":
    asyncio.run(patrol())
